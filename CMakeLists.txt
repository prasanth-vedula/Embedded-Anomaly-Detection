cmake_minimum_required(VERSION 3.16)
project(embedded_anomaly_detection C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -Wstrict-prototypes -Wshadow)
include_directories(Inc)
set(SOURCES Src/HAL/hal.c Src/Sensor/sensor.c Src/Preprocess/preprocess.c Src/Window/window.c Src/TinyML/tinyml.c Src/Anomaly/anomaly.c Src/Event/event.c Src/Telemetry/telemetry.c)
add_library(anomaly_core ${SOURCES})
target_link_libraries(anomaly_core m)
add_executable(anomaly_demo Src/main.c)
target_link_libraries(anomaly_demo anomaly_core m)
enable_testing()
add_executable(test_embedded ../tests/embedded/test_embedded.c)
target_compile_options(test_embedded PRIVATE -UNDEBUG)
target_link_libraries(test_embedded anomaly_core m)
add_test(NAME embedded_unit COMMAND test_embedded)
